Step 1 — Google Apps Script Setup One-time setup
Kya yeh karta hai? Google Apps Script ek free backend hai jo Google Drive mein aapka ERP data automatically save aur load karta hai. Browser clear ho, naya device ho — data hamesha Google Drive mein mehfooz rahega.
1

Naya script banao — yahan click karein:
script.google.com/create
2
Neeche ka code copy karein aur Script Editor mein paste karein
(Code.gs file mein jo kuch hai sab delete karo, phir yeh paste karo)
// Gold ERP — Google Apps Script Backend
// Banaya gaya: Gold ERP v13

const FOLDER_NAME = 'Gold_ERP_Data';
const FILE_NAME   = 'goldErpData.json';

function doGet(e){
  try{
    const file = _getOrCreateFile();
    const text = file.getBlob().getDataAsString();
    return ContentService
      .createTextOutput(text || '{}')
      .setMimeType(ContentService.MimeType.JSON);
  }catch(err){
    return ContentService
      .createTextOutput(JSON.stringify({error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e){
  try{
    const data = e.postData.contents;
    const file = _getOrCreateFile();
    file.setContent(data);
    return ContentService
      .createTextOutput(JSON.stringify({ok:true, ts: new Date().toISOString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }catch(err){
    return ContentService
      .createTextOutput(JSON.stringify({error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function _getOrCreateFile(){
  const folders = DriveApp.getFoldersByName(FOLDER_NAME);
  const folder  = folders.hasNext()
    ? folders.next()
    : DriveApp.createFolder(FOLDER_NAME);
  
  const files = folder.getFilesByName(FILE_NAME);
  if(files.hasNext()) return files.next();
  
  return folder.createFile(FILE_NAME, '{}', MimeType.PLAIN_TEXT);
}
